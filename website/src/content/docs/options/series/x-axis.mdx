The x axis the series is drawn against, named by the axis's id.

<OptionFacts series="xAxis" />

## What it does

Axis ids are given by [`id`](../../axis/id/) on the axis, or are `x`, then `x2`, `x3`… by position; x and y axes share
one set of ids. An id no x axis has is reported as `unknown-axis`, with the id as detail, and the series goes on the
first x axis. On a bar chart the x axis runs upright, but it is still the series' `xAxis`.

## Example

<CodeTabs>
<Fragment slot="ts">

```ts
import { column } from '@dzgraph/core';
import { chart } from '@dzgraph/dom';

chart(
  document.getElementById('honey')!,
  {
    xAxis: { id: 'months', scale: 'category', categories: ['Jan', 'Feb', 'Mar'] },
    series: [{ type: 'column', id: 'honey', name: 'Honey (kg)', xAxis: 'months', data: [14, 22, 19] }],
  },
  { types: [column] },
);
```

</Fragment>
<Fragment slot="angular">

```ts
import { Component } from '@angular/core';
import { column, type ChartOptions } from '@dzgraph/core';
import { DzChart } from '@dzgraph/angular';

@Component({
  selector: 'app-honey',
  imports: [DzChart],
  template: `<dz-chart style="block-size: 300px" [options]="options" [types]="types" />`,
})
export class Honey {
  readonly types = [column];
  readonly options: ChartOptions = {
    xAxis: { id: 'months', scale: 'category', categories: ['Jan', 'Feb', 'Mar'] },
    series: [{ type: 'column', id: 'honey', name: 'Honey (kg)', xAxis: 'months', data: [14, 22, 19] }],
  };
}
```
