## Chart
  export interface Chart {
      update(changes: ChartUpdate): void;
      setData(seriesId: string, data: SeriesOptions['data']): void;
      setSeriesVisible(seriesId: string, visible: boolean): void;
      setRange(range: {
          readonly min: number | string;
          readonly max: number | string;
      } | null): void;
      getView(): ChartView;
      setView(view: ChartView): void;
      showX(x: number | null): void;
      selectBy(by: 'key' | 'x', values: readonly (string | number)[]): void;
      setMode(mode: Mode): void;
      exportSvg(options?: ExportOptions): string | null;
      exportPng(options?: PngOptions): Promise<Blob | null>;
      exportCsv(options?: {
          readonly locale?: boolean;
      }): string | null;
      download(format: ExportFormat, options?: PngOptions): Promise<void>;
      print(options?: {
          readonly table?: boolean;
      }): void;
      readonly mode: Mode;
      readonly problems: readonly Problem[];
      on<E extends keyof ChartEvents>(event: E, handler: (detail: ChartEvents[E]) => void): () => void;
      destroy(): void;
  }

  }

## Feature
  export interface Feature {
      readonly name: 'timeAxis' | 'zoom' | 'navigatorStrip' | 'canvas' | 'webgl' | 'crosshair' | 'compare' | 'measure' | 'selection' | 'aggregation' | 'drilldown' | 'menu' | 'exporting' | 'tooltip';
      readonly needs?: readonly Feature[];
      readonly time?: TimeKit;
      readonly navigator?: {
          readonly kit: NavigatorKit;
          readonly controls: (svg: SVGSVGElement, group: SVGGElement, root: HTMLElement, hooks: NavigatorPartHooks) => NavigatorControlsPart;
      };
      readonly zoom?: (root: HTMLElement, svg: SVGSVGElement, hooks: ZoomHooks) => ZoomPart;
      readonly canvas?: (root: HTMLElement) => Painter & {
          destroy(): void;
      };
      readonly webgl?: (root: HTMLElement, changed: () => void) => WebglPart;
      readonly compare?: CompareKit;
      readonly aggregation?: AggregationKit;
      readonly drill?: (hooks: DrillHooks) => DrillPart;
      readonly menu?: (root: HTMLElement, hooks: MenuHooks) => MenuPart;
      readonly exporting?: ExportingKit;
      readonly tooltip?: (root: HTMLElement) => TooltipDrawer;
      readonly measure?: (svg: SVGSVGElement, hooks: MeasureHooks) => MeasurePart;
      readonly selection?: {
          readonly kit: SelectionKit;
          readonly part: (svg: SVGSVGElement, hooks: SelectionHooks) => SelectionPart;
      };
      readonly modeBar?: (root: HTMLElement, hooks: ModeBarHooks) => ModeBar;
      readonly crosshair?: {
          readonly kit: CrosshairKit;
          readonly part: (svg: SVGSVGElement, hooks: CrosshairHooks) => CrosshairPart;
      };
  }
