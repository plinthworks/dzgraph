  export type GridFilterLike = {
      readonly type: 'set';
      readonly include: readonly string[];
  } | {
      readonly type: 'number';

## GridReadLike
  export interface GridReadLike {
      readonly kernel: {
          readonly rows: {
              readonly displayed: {

          };
          readonly columns: {
              visible(): readonly GridColumnLike[];
          };
          readonly selection: GridSelectionLike;
          readonly reader: {
              value(col: GridColumnLike, row: unknown, id: string): unknown;

              readonly type: string;
              readonly source?: string;

      };
  }
